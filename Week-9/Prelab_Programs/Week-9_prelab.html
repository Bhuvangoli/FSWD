1. Math Module

Code:

Math.js

function add(a, b) {
    return a + b;
}

function subtract(a, b) {
    return a - b;
}

function multiply(a, b) {
    return a * b;
}

function divide(a, b) {
    if (b === 0) {
        return "Cannot divide by zero";
    }
    return a / b;
}

module.exports = {
    add,
    subtract,
    multiply,
    divide
};

Main.js

const math = require('./math.js');

console.log("Addition:", math.add(10, 5));
console.log("Subtraction:", math.subtract(10, 5));
console.log("Multiplication:", math.multiply(10, 5));
console.log("Division:", math.divide(10, 5));

Output:

Addition: 15
Subtraction: 5
Multiplication: 50
Division: 2

2. Read a text file and handle errors

Code:

const fs = require('fs');

const filename = process.argv[2];

if (!filename) {
    console.log("Please provide a filename.");
    process.exit(1);
}

fs.readFile(filename, 'utf8', (err, data) => {
    if (err) {
        console.log("Error: File does not exist or cannot be read.");
        return;
    }

    console.log("File Content:");
    console.log(data);
});

Output:

File Content:
Welcome to Node.js!
This is a sample text file.
File handling using Node.js.

3. Read JSON

Code:

const fs = require('fs');
fs.readFile('users.json', 'utf8', (err, data) => {

    if (err) {
        console.log("Error reading JSON file:", err.message);
        return;
    }

    try {
        const users = JSON.parse(data);

        users.forEach((user) => {
            console.log("Name:", user.name);
            console.log("Age:", user.age);
            console.log("Email:", user.email);
            console.log("-------------------");
        });

    } catch (error) {
        console.log("Invalid JSON format.");
    }
});

4. Simple REST API

Code:

const express = require('express');

const app = express();

app.use(express.json());

let users = [
    { id: 1, name: "John", age: 22 },
    { id: 2, name: "Alice", age: 25 },
    { id: 3, name: "David", age: 21 }
];

app.get('/users', (req, res) => {
    res.json(users);
});

app.post('/users', (req, res) => {
    const newUser = {
        id: users.length > 0
            ? Math.max(...users.map(u => u.id)) + 1
            : 1,
        name: req.body.name,
        age: req.body.age
    };

    users.push(newUser);

    res.status(201).json({
        message: "User added successfully",
        user: newUser
    });
});

app.delete('/users/:id', (req, res) => {
    const id = parseInt(req.params.id);

    const userExists = users.some(user => user.id === id);

    if (!userExists) {
        return res.status(404).json({
            message: "User not found"
        });
    }

    users = users.filter(user => user.id !== id);

    res.json({
        message: "User deleted successfully"
    });
});

app.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});

Output:

Server running at http://localhost:3000
