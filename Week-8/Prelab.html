1. Server Date and Time

Code

const http = require('http');

const server = http.createServer((req, res) => {
    res.writeHead(200, { 'Content-Type': 'text/plain' });
    res.end('Current Date and Time: ' + new Date());
});

server.listen(3000, () => {
    console.log('Server running at http://localhost:3000');
});

Output

Current Date and Time: Mon Oct 12 2026 19:47:31 GMT+0530 (India Standard Time)

Example output from the provided program. The date and time will change each time the server is requested.

2. Upper Case

Code

const fs = require('fs');

fs.readFile('input.txt', 'utf8', (err, data) => {
    if (err) {
        console.log(err);
        return;
    }

    const upperData = data.toUpperCase();

    fs.writeFile('output.txt', upperData, (err) => {
        if (err) {
            console.log(err);
            return;
        }

        console.log('Content written in uppercase successfully!');
    });
});

Output

Content written in uppercase successfully!

THIS LINE CONTAINS THE TEXT IN LOWERCASE AND CONVERTED TO UPPERCASE

3. Callback Hell

Code

console.log("Callback Hell:");

setTimeout(() => {
    console.log("First task");

    setTimeout(() => {
        console.log("Second task");

        setTimeout(() => {
            console.log("Third task");

            runTasks();
        }, 1000);

    }, 1000);
}, 1000);


function task(message) {
    return new Promise((resolve) => {
        setTimeout(() => {
            console.log(message);
            resolve();
        }, 1000);
    });
}

async function runTasks() {
    console.log("\\nUsing Async/Await:");

    await task("First task");
    await task("Second task");
    await task("Third task");
}

Output

Callback Hell:
First task
Second task
Third task

Using Async/Await:
First task
Second task
Third task

4. Custom Event

Code

5. Sync and Async File Reading

Code

const fs = require('fs');

console.log('Synchronous File Reading:');

try {
    const data = fs.readFileSync('input.txt', 'utf8');
    console.log('File content:');
    console.log(data);
    console.log('Synchronous reading completed.\\n');
} catch (err) {
    console.log(err.message);
}

console.log('Asynchronous File Reading:');

fs.readFile('input.txt', 'utf8', (err, data) => {
    if (err) {
        console.log(err.message);
        return;
    }

    console.log('File content:');
    console.log(data);
    console.log('Asynchronous reading completed.');
});

console.log('Reading started.');

Output

Synchronous File Reading:
File content:
this line contains the text in lowercase and converted to uppercase
Synchronous reading completed.

Asynchronous File Reading:
Reading started.
File content:
this line contains the text in lowercase and converted to uppercase
Asynchronous reading completed.
