4. Reverse Each Word in a String



function reverseWords(str) {
    let words = str.split(" ");

    let result = words.map(function(word) {
        return word.split("").reverse().join("");
    });

    return result.join(" ");
}

5. Find Second Smallest and Second Largest Numbers



function secondSmallestLargest(arr) {
    let unique = [...new Set(arr)];

    unique.sort(function(a, b) {
        return a - b;
    });

    let secondSmallest = unique[1];
    let secondLargest = unique[unique.length - 2];

    return "Second Smallest: " + secondSmallest +
           "\nSecond Largest: " + secondLargest;
}

6. Count Occurrences of Each Letter



function countLetters(str) {
    let count = {};

    for (let char of str) {
        if (count[char]) {
            count[char]++;
        } else {
            count[char] = 1;
        }
    }

    let output = "";

    for (let char in count) {
        output += char + ":" + count[char] + ", ";
    }

    return output;
}

7. Generate All Possible Substrings



function substrings(str) {
    let result = [];

    for (let i = 0; i < str.length; i++) {
        for (let j = i + 1; j <= str.length; j++) {
            result.push(str.substring(i, j));
        }
    }

    return result;
}

8. Find the First Non-Repeated Character



function firstNonRepeated(str) {
    let count = {};

    for (let char of str) {
        if (count[char]) {
            count[char]++;
        } else {
            count[char] = 1;
        }
    }

    for (let char of str) {
        if (count[char] === 1) {
            return char;
        }
    }

    return "No non-repeated character";
}

9. Check Password Strength



function checkPassword(password) {
    let pattern =
        /^(?=.*[a-z])(?=.*[A-Z])(?=.*[0-9])(?=.*[@#$%^&*!]).{8,}$/;

    if (pattern.test(password)) {
        return "Strong Password";
    } else {
        return "Not a Strong Password";
    }
}

10. Remove Duplicate Characters



function removeDuplicates(str) {
    let result = "";

    for (let char of str) {
        if (!result.includes(char)) {
            result += char;
        }
    }

    return result;
}
