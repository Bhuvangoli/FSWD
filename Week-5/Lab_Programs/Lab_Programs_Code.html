1. Mobile Number Validation

Code:

const readline = require("readline");

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});
rl.question("Enter your mobile number: ", (mobile) => {

    let pattern = /^[6-9][0-9]{9}$/;

    if (pattern.test(mobile)) {
        console.log("Valid mobile number");
    } else {
        console.log("Invalid mobile number");
    }

    rl.close();
});

Output:

Enter your mobile number: 9876543210 Valid mobile number

2. Arithmetic Operations

Code:

const readline = require("readline");

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

function calculate(a, b, operator, callback) {
    let result;

    if (operator === "+") {
        result = a + b;
    }
    else if (operator === "-") {
        result = a - b;
    }
    else if (operator === "*") {
        result = a * b;
    }
    else if (operator === "/") {
        if (b === 0) {
            result = "Cannot divide by zero";
        } else {
            result = a / b;
        }
    }
    else {
        result = "Invalid operator";
    }

    callback(result);
}

function display(result) {
    console.log("Result:", result);
}

rl.question("Enter first number: ", (input1) => {
    rl.question("Enter second number: ", (input2) => {
        rl.question("Enter operator (+, -, *, /): ", (operator) => {
            let a = Number(input1);
            let b = Number(input2);
            calculate(a, b, operator, display);
            rl.close();
        });
    });
});

Output:

Enter first number: 1 Enter second number: 2 Enter operator (+, -, *, /): * Result: 2

6. Arrow Function

Code:

const students = [

    { name: "A", marks: 35 },

    { name: "B", marks: 78 },

    { name: "C", marks: 42 },

    { name: "D", marks: 90 }

];

const marks = students.map(student => student.marks);

const total = marks.reduce((sum, mark) => sum + mark, 0);

const average = total / students.length;

const passedStudents = students.filter(student => student.marks >= 40);

console.log("Marks:", marks);

console.log("Class Average:", average);

console.log("Passed Students:");

passedStudents.forEach(student => {

    console.log(student.name + " - " + student.marks);

});

Output:

Marks: [ 35, 78, 42, 90 ] Class Average: 61.25 Passed Students: B - 78 C - 42 D - 90

7. Traffic Lights

Code:

<!DOCTYPE html>
<html>
<head>
    <title>Traffic Light System</title>

    <style>
        #light {
            width: 100px;
            height: 100px;
            border-radius: 50%;
            background-color: gray;
            margin: 20px;
        }

        button {
            padding: 10px 20px;
            margin: 5px;
        }
    </style>
</head>

<body>

<h2>Traffic Light System</h2>

<div id="light"></div>

<button id="start">Start</button>
<button id="stop">Stop</button>

<p id="message">Traffic light stopped.</p>

<script>

const light = document.getElementById("light");
const message = document.getElementById("message");

let running = false;
let timer;

function changeLight(color, next) {

    if (!running) {
        return;
    }

    light.style.backgroundColor = color;
    message.innerHTML = "Current Light: " + color;

    timer = setTimeout(next, 2000);
}

function red() {
    changeLight("red", green);
}

function green() {
    changeLight("green", yellow);
}

function yellow() {
    changeLight("yellow", red);
}

document.getElementById("start").addEventListener("click", () => {

    if (!running) {
        running = true;
        message.innerHTML = "Traffic light started.";
        red();
    }

});

document.getElementById("stop").addEventListener("click", () => {

    running = false;
    clearTimeout(timer);

    light.style.backgroundColor = "gray";
    message.innerHTML = "Traffic light stopped.";

});

</script>

</body>
</html>

Output:

Traffic Light System

Current Light: yellow
