1. Square and Cube of Number

Code:

const square = (n) => n * n;

const cube = (n) => n * n * n;

let num = 5;

console.log("Number:", num);

console.log("Square:", square(num));

console.log("Cube:", cube(num));

Output:

Number: 5 Square: 25 Cube: 125

2. Add Two Numbers Using Callback

Code:

function add(a, b, callback) {

    let result = a + b;

    callback(result);

}

function display(result) {

    console.log("Sum:", result);

}

add(10, 20, display);

Output:

Sum: 30

3. Promise Resolving

Code:


function checkNumber(num) {

    return new Promise((resolve, reject) {

        if (num < 0) {

            reject("Number is negative!");

        } else {

            setTimeout(() => {

                resolve("Success! Number is valid.");

            }, 2000);

        }

    });

}

checkNumber(10)

    .then((message) => {

        console.log(message);

    })

    .catch((error) => {

        console.log(error);

    });

Output:

Success! Number is valid.

4. Async/Await

Code:


function fetchData() {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("Data fetched successfully!");
        }, 3000);
    });
}

async function getData() {
    console.log("Fetching data...");

    const data = await fetchData();

    console.log(data);
}

getData();

Output:

Fetching data... Data fetched successfully!

5. Email Address Validation

Code:

function validateEmail(email) {

    const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

    if (pattern.test(email)) {

        console.log("Valid email address");

    } else {

        console.log("Invalid email address");

    }

}

let email = "student@gmail.com";

validateEmail(email);

Output:

Valid email address
